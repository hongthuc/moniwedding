        // Xử lý Modal Chi Tiết Sản Phẩm
        function openProductModal(productId) {
            const item = REAL_PRODUCTS.find(p => p.id === productId);
            if (!item) return;
            currentModalProduct = item;
            document.getElementById('modal-category').textContent = item.categoryName;
            document.getElementById('modal-title').textContent = item.title;
            document.getElementById('modal-desc').textContent = item.desc;
            const modalImg = document.getElementById('modal-img');
            modalImg.setAttribute('data-drive-id', item.driveId);
            modalImg.removeAttribute('data-tried-lh3');
            modalImg.src = 'https://drive.google.com/thumbnail?id=' + item.driveId + '&sz=w1200';
            modalImg.onerror = function() {
                handleDriveImageError(modalImg, item.fallback);
            };
            document.getElementById('product-modal').classList.remove('hidden');
        }

                icon.className = 'fa-solid fa-xmark text-xl';
            } else {
                panel.classList.add('hidden');
                icon.className = 'fa-solid fa-bars text-lg';
            }
        }
        function closeMobileMenu() {
            document.getElementById('mobile-menu-panel').classList.add('hidden');
            document.getElementById('mobile-menu-icon').className = 'fa-solid fa-bars text-lg';
        }

        // Xử lý 2 Banner Tự Trượt Chuyên Nghiệp
        let currentBannerIndex = 0;
        const totalBanners = 2;
        let bannerInterval = null;

        function showBannerSlide(index) {
            currentBannerIndex = (index + totalBanners) % totalBanners;
            const slides = document.querySelectorAll('.hero-slide');
            slides.forEach((slide, i) => {
                if (i === currentBannerIndex) {
                    slide.classList.remove('opacity-0', 'z-0', 'pointer-events-none');
                    slide.classList.add('opacity-100', 'z-10');
                } else {
                    slide.classList.remove('opacity-100', 'z-10');
                    slide.classList.add('opacity-0', 'z-0', 'pointer-events-none');
                }
            });

            const dot0 = document.getElementById('slider-dot-0');
            const dot1 = document.getElementById('slider-dot-1');
            if (dot0 && dot1) {
                if (currentBannerIndex === 0) {
                    dot0.className = 'w-8 h-2 rounded-full bg-moni-gold transition-all';
                    dot1.className = 'w-2.5 h-2 rounded-full bg-white/60 hover:bg-white transition-all';
                } else {
                    dot1.className = 'w-8 h-2 rounded-full bg-moni-gold transition-all';
                    dot0.className = 'w-2.5 h-2 rounded-full bg-white/60 hover:bg-white transition-all';
                }
            }
        }

        function changeBannerSlide(step) {
            showBannerSlide(currentBannerIndex + step);
            resetBannerTimer();
        }

        function goToBannerSlide(index) {
            showBannerSlide(index);
            resetBannerTimer();
        }

        function resetBannerTimer() {
            if (bannerInterval) clearInterval(bannerInterval);
            bannerInterval = setInterval(() => {
                showBannerSlide(currentBannerIndex + 1);
            }, 5000);
        }

        // Khởi chạy khi tải trang
        window.addEventListener('DOMContentLoaded', () => {
            renderProducts('all');
            renderPosts('all');
            resetBannerTimer();
        });
